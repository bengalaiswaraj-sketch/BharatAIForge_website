agent_rag_pipeline.py — Guided Lab Preview 
from fastapi import FastAPI
from bharatai_forge.rag import HybridRetriever, CrossEncoderReranker
from bharatai_forge.agents import AgentOrchestrator, GuardrailEvaluator

app = FastAPI(title="BharatAIForge-Production-Service")
retriever = HybridRetriever(collection="campus_knowledge_v2", top_k=8)
reranker = CrossEncoderReranker(top_n=3)
guardrails = GuardrailEvaluator(min_faithfulness=0.92)

@app.post("/v1/agent/query")
async def execute_grounded_agent(query: str):
    docs = reranker.rank(query, await retriever.search(query))
    response = await AgentOrchestrator.run_with_mcp(query=query, context=docs)
    audit = guardrails.verify(query=query, context=docs, answer=response)
    return {"answer": response, "citations": docs.sources, "audit": audit}
